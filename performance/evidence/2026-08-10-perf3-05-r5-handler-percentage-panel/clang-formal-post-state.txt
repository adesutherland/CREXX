path='/private/tmp/crexx-perf3-05-r5.9gYg9z/clang-formal-timing/summary.csv'
rows=CSV.read(path,headers:true)
panels=%w[profile-15 profile-20 profile-30]
engines=%w[rxtvm rxbvm]
workloads=%w[Sieve Permute Bounce Richards Base64 Towers RexxCPS]
values={}
rows.each do |row|
  engine=row['vm'].start_with?('rxtvm') ? 'rxtvm' : 'rxbvm'
  panel=row['vm'].sub(/^#{engine}-clang-r5-/,'')
  values[[row['workload'],engine,panel]]=row['median'].to_f
end
puts 'panel,engine,all7_pct,without_base64_pct,common5_pct,worst_workload,worst_pct,guard_cells'
engines.each do |engine|
  panels.each do |panel|
    ratios={}; deltas={}
    workloads.each do |workload|
      ratio=values[[workload,engine,'all-inline']]/values[[workload,engine,panel]]
      ratios[workload]=ratio; deltas[workload]=(ratio-1)*100
    end
    gm=lambda{|set|Math.exp(set.sum{|w|Math.log(ratios[w])}/set.size)-1}
    worst=deltas.min_by{|_,v|v}; guards=deltas.select{|_,v|v < -3}.keys
    guards << 'COMMON_GM' if gm.call(%w[Sieve Permute Bounce Richards Base64])*100 < -1
    printf "%s,%s,%+.3f,%+.3f,%+.3f,%s,%+.3f,%s\n",panel,engine,gm.call(workloads)*100,gm.call(workloads-[%q{Base64}])*100,gm.call(%w[Sieve Permute Bounce Richards Base64])*100,worst[0],worst[1],guards.join('+')
  end
end
puts
puts 'workload,engine,'+panels.join(',')
workloads.each do |workload|
  engines.each do |engine|
    values_out=panels.map{|panel|(values[[workload,engine,'all-inline']]/values[[workload,engine,panel]]-1)*100}
    puts ([workload,engine]+values_out.map{|v|format('%+.3f',v)}).join(',')
  end
end
RUBY
