cmake_minimum_required(VERSION 3.24)
project(LevelLBootstrap NONE)
# This standalone graph intentionally cannot construct re2c or Level G. It uses
# the user's already-built, unchanged C tools and Level B prerequisite image.
set(CREXX_LEVEL_L_STANDALONE ON)
set(CREXX_HOST_BIN "" CACHE PATH "Existing native rxc/rxas/rxlink/VM and library.rxbin directory")
option(CREXX_LEVEL_L_NOOPT "Build Level L with compiler and assembler optimization disabled" OFF)
get_filename_component(CREXX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${CREXX_ROOT}/cmake/CrexxImportRoots.cmake")
set(_host_tools rxc rxas rxlink rxbvm)
if(EXISTS "${CREXX_HOST_BIN}/rxtvm${CMAKE_EXECUTABLE_SUFFIX}")
  list(APPEND _host_tools rxtvm)
endif()
foreach(tool IN LISTS _host_tools)
  if(NOT EXISTS "${CREXX_HOST_BIN}/${tool}${CMAKE_EXECUTABLE_SUFFIX}")
    message(FATAL_ERROR "Missing native prerequisite ${CREXX_HOST_BIN}/${tool}${CMAKE_EXECUTABLE_SUFFIX}")
  endif()
  add_executable(${tool} IMPORTED GLOBAL)
  set_target_properties(${tool} PROPERTIES IMPORTED_LOCATION "${CREXX_HOST_BIN}/${tool}${CMAKE_EXECUTABLE_SUFFIX}")
endforeach()
if(NOT EXISTS "${CREXX_HOST_BIN}/library.rxbin")
  message(FATAL_ERROR "Missing existing Level B library.rxbin prerequisite")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/bin/library.rxbin"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CREXX_HOST_BIN}/library.rxbin" "${CMAKE_BINARY_DIR}/bin/library.rxbin"
  DEPENDS "${CREXX_HOST_BIN}/library.rxbin" VERBATIM)
add_custom_target(library DEPENDS "${CMAKE_BINARY_DIR}/bin/library.rxbin")
add_subdirectory("${CREXX_ROOT}/lib/rxfnsl/rexx" rexx)
add_subdirectory("${CREXX_ROOT}/langtools" langtools)
include("${CREXX_ROOT}/tools/unicode/LevelL.cmake")
